<script>
(() => {
  'use strict';
  const STORE_KEY = 'mcf_pro_v4';
  const $ = id => document.getElementById(id);
  
  let state = { moves: [], debts: [], payments: [], cards: [] };

  function init() {
    loadData();
    setupEvents();
    render();
    $('today').textContent = new Date().toLocaleDateString('es-MX', { weekday: 'short', day: 'numeric', month: 'short' });
  }

  function loadData() {
    try { state = JSON.parse(localStorage.getItem(STORE_KEY)) || state; } catch (e) { console.error(e); }
  }
  function saveData() {
    localStorage.setItem(STORE_KEY, JSON.stringify(state));
    render();
  }

  function setupEvents() {
    // Navigation
    document.querySelectorAll('#tabs button').forEach(btn => {
      btn.onclick = () => {
        document.querySelectorAll('#tabs button').forEach(b => b.classList.remove('active'));
        document.querySelectorAll('.pane').forEach(p => p.classList.add('hidden'));
        btn.classList.add('active');
        $(btn.dataset.tab).classList.remove('hidden');
      };
    });

    // Form Submissions
    $('moveForm').onsubmit = (e) => {
      e.preventDefault();
      const move = {
        id: Date.now().toString(),
        type: $('moveType').value,
        space: $('moveSpace').value,
        concept: $('moveConcept').value,
        amount: parseFloat($('moveAmount').value),
        date: $('moveDate').value,
        category: $('moveCategory').value,
        tags: $('moveTags').value.split(',').map(t => t.trim()).filter(Boolean)
      };
      state.moves.push(move);
      saveData();
      $('moveForm').reset();
    };

    // Calculator
    const updateCalc = () => {
      const amt = parseFloat($('calcAmount').value) || 0;
      const rate = (parseFloat($('calcRate').value) || 0) / 100;
      const fixed = parseFloat($('calcFixed').value) || 0;
      const total = (amt + fixed) / (1 - rate);
      $('calcResult').textContent = `$${total.toFixed(2)}`;
    };
    ['calcAmount', 'calcRate', 'calcFixed'].forEach(id => $(id).oninput = updateCalc);
    updateCalc();

    // App Shell & Auth
    $('authForm').onsubmit = (e) => {
      e.preventDefault();
      $('authScreen').classList.add('hidden');
      $('appShell').classList.remove('hidden');
    };

    $('printReportBtn').onclick = () => window.print();
  }

  function render() {
    // Metrics calculation
    const income = state.moves.filter(m => m.type === 'income').reduce((acc, m) => acc + m.amount, 0);
    const expense = state.moves.filter(m => m.type === 'expense').reduce((acc, m) => acc + m.amount, 0);
    const balance = income - expense;

    $('metricBalance').textContent = `$${balance.toFixed(2)}`;
    $('metricIncome').textContent = `$${income.toFixed(2)}`;
    $('metricExpense').textContent = `$${expense.toFixed(2)}`;

    renderSVGChart(income, expense);
    renderMovementsTable();
  }

  // Renderizador de Gráfico Visual SVG Nativo
  function renderSVGChart(inc, exp) {
    const max = Math.max(inc, exp, 1);
    const incHeight = (inc / max) * 160;
    const expHeight = (exp / max) * 160;

    $('svgChartWrap').innerHTML = `
      <svg width="100%" height="100%" viewBox="0 0 300 200" preserveAspectRatio="none">
        <line x1="10" y1="180" x2="290" y2="180" stroke="var(--line)" stroke-width="2"/>
        <!-- Bar Ingresos -->
        <rect class="chart-bar" x="60" y="${180 - incHeight}" width="50" height="${incHeight}" fill="var(--green)" rx="6"/>
        <text x="85" y="${170 - incHeight}" text-anchor="middle" font-size="12" fill="var(--ink)">+$${inc.toFixed(0)}</text>
        <text x="85" y="195" text-anchor="middle" font-size="11" fill="var(--muted)">Ingresos</text>
        
        <!-- Bar Gastos -->
        <rect class="chart-bar" x="190" y="${180 - expHeight}" width="50" height="${expHeight}" fill="var(--red)" rx="6"/>
        <text x="215" y="${170 - expHeight}" text-anchor="middle" font-size="12" fill="var(--ink)">-$${exp.toFixed(0)}</text>
        <text x="215" y="195" text-anchor="middle" font-size="11" fill="var(--muted)">Gastos</text>
      </svg>
    `;
  }

  function renderMovementsTable() {
    const rows = state.moves.map(m => `
      <tr>
        <td>${m.date}</td>
        <td><b>${m.concept}</b> ${m.tags ? m.tags.map(t => `<span class="pill">${t}</span>`).join(' ') : ''}</td>
        <td>${m.category}</td>
        <td class="${m.type === 'income' ? 'green' : 'red'}" style="font-weight:bold;">
          ${m.type === 'income' ? '+' : '-'}$${m.amount.toFixed(2)}
        </td>
      </tr>
    `).join('');
    $('movesTable').innerHTML = rows ? `<table><thead><tr><th>Fecha</th><th>Concepto</th><th>Categoría</th><th>Monto</th></tr></thead><tbody>${rows}</tbody></table>` : `<p class="muted" style="text-align:center; padding: 20px;">No hay registros aún.</p>`;
  }

  init();
})();
</script>
